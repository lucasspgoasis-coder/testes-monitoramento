Conexão com Google Planilhas

Sincronização imediata de múltiplos testes com fotos indicadas

Para que funcione perfeitamente, configure a primeira linha (A1 a H1) com estes nomes:
Data | Semana | Bloco | Teste | Quantidade | Unidade | Mudas | Resultado | Foto

Se o bloco tiver mais de um teste associado, o sistema gravará automaticamente uma linha correspondente para cada teste na sua planilha.

Código atualizado para o Google Apps Script 
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Abra o script diretamente pela planilha."}))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader("Access-Control-Allow-Origin", "*");
    }
    
    var sheet = ss.getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    if (Array.isArray(data)) {
      data.forEach(function(row) {
        sheet.appendRow([
          row.Data,
          row.Semana,
          row.Bloco,
          row.Teste,
          row.Quantidade,
          row.Unidade,
          row.Mudas,
          row.Resultado,
          row.Foto
        ]);
      });
    } else {
      sheet.appendRow([
        data.Data,
        data.Semana,
        data.Bloco,
        data.Teste,
        data.Quantidade,
        data.Unidade,
        data.Mudas,
        data.Resultado,
        data.Foto
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader("Access-Control-Allow-Origin", "*");
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader("Access-Control-Allow-Origin", "*");
  }
}

function doOptions(e) {
  return ContentService.createTextOutput("")
    .setHeader("Access-Control-Allow-Origin", "*")
    .setHeader("Access-Control-Allow-Methods", "POST, OPTIONS")
    .setHeader("Access-Control-Allow-Headers", "Content-Type");
}
